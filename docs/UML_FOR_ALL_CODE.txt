UML / Class and Module Overview
===============================

Frontend Classes
----------------
class AuthService {
  currentUser()
  login(email, password)
  register(data)
  logout()
}

class ApiService {
  login(email, password)
  register(payload)
  getExams()
  getExam(id)
  createExam(payload)
  updateExam(id, patch)
  deleteExam(id)
  getSubmissions()
  submitExam(examId, answers)
  updateSubmissionGrade(id, payload)
  getTeacherAnalytics()
}

class StorageService {
  get(key, fallback)
  set(key, value)
  remove(key)
  clearProjectData()
}
